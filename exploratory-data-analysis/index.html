Exploratory Data Analysis (EDA)

EDA is the foundation of every ML project — you can't build an accurate model without first understanding missing values, distributions, and correlations in your data.

Missing Data Analysis

def analyze_missing_data(dataframe):
    missing_percent = (dataframe.isnull().sum() / dataframe.shape[0] * 100).sort_values(ascending=False)
    missing_percent = missing_percent.apply(lambda x: f"{x:.2f}%")
    info_df = pd.DataFrame({'DataType': dataframe.dtypes, 'Null Count': dataframe.isnull().sum()})
    combined_df = pd.concat([missing_percent, info_df], axis=1)
    combined_df.columns = ['Missing Percent', 'DataType', 'Null Count']
    return combined_df.sort_values(by='Missing Percent', ascending=False)

Correlation Analysis

python
import seaborn as sns
import matplotlib.pyplot as plt

def analyze_correlations(df, target_column="SalePrice", corr_threshold=0.7):
    numeric_df = df.select_dtypes(include=['number'])
    correlation_matrix = numeric_df.corr()

    top_corr = correlation_matrix.index[abs(correlation_matrix[target_column]) > 0.5]
    plt.figure(figsize=(10, 8))
    sns.heatmap(numeric_df[top_corr].corr(), annot=True, cmap="YlGnBu", square=True)
    plt.show()

    abs_corr = correlation_matrix.abs()
    pairs = abs_corr.stack().reset_index()
    pairs.columns = ['Variable1', 'Variable2', 'Correlation']
    pairs = pairs[pairs['Variable1'] < pairs['Variable2']]
    return pairs[pairs['Correlation'] >= corr_threshold].sort_values('Correlation', ascending=False)

Feature Distribution Visualization

python
def visualize_feature(df, feature, plot_type="box", target_column="SalePrice"):
    """plot_type: 'box', 'kde', 'hist', 'violin', or 'scatter'"""
    plt.figure(figsize=(10, 6))
    if plot_type == "box":
        sns.boxplot(x=df[feature])
    elif plot_type == "kde":
        sns.kdeplot(data=df[feature], fill=True)
    elif plot_type == "hist":
        sns.histplot(data=df[feature], bins=30, kde=True)
    elif plot_type == "scatter":
        sns.scatterplot(data=df, x=feature, y=target_column)
    plt.tight_layout()
    plt.show()

# Usage:
# visualize_feature(train_df, 'GrLivArea', plot_type='scatter', target_column='SalePrice')
A note on outliers

Scatter plots against the target are a fast way to spot potential outliers visually. For a full detection and treatment workflow, see the Data Preprocessing page.
